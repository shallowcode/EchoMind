```python
def render_tasks(tasks):
    """No freeform rewriting after validation: one section per task, including failures."""
    return "\n\n".join(
        f"【{t.title}】\n{t.answer}"
        + (
            "\n尚未完成：" + "；".join(t.missing)
            if t.status not in {"completed", "denied"}
            else ""
        )
        for t in tasks
    )

```
